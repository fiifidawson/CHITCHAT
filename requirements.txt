# CHITCHAT pipeline dependencies.
#
# Hand-curated DIRECT dependencies only. This replaces a raw `pip freeze` that
# did not install a working environment: it pinned `pathlib==1.0.1` (a PyPI
# backport that shadows the stdlib module and breaks on Python 3), the
# `bs4==0.0.2` dummy package, a `configparser` backport that shadows the
# stdlib, and an unrelated neuroimaging stack (nipype, nibabel, pyxnat, prov,
# traits), while omitting six packages the code actually imports.
#
# Transitive dependencies are resolved by pip and deliberately not listed here.
# Every entry below is imported somewhere in the tree; the comments say where.
#
#     pip install -r requirements.txt
#
# The documentation toolchain is NOT here. It lives in the `docs` dependency
# group in pyproject.toml -- see docs/project/contributing.md.
#
# Resolves and installs on Python 3.10 (.python-version). On 3.10 the resolver
# selects the last 3.10-compatible pandas/numpy/matplotlib; moving to 3.12+
# would pick up the current major releases.

# --- Screening -- legacy/src/screen_papers.py -------------------------------
# Uses client.beta.chat.completions.parse(...) with a pydantic response_format.
# That call still exists and accepts the same arguments in openai 3.x, so the
# cap is the next major rather than <2.
openai>=1.98,<4
pydantic>=2.11,<3
tqdm>=4.67,<5

# --- Discovery and scraping -- legacy/src/api/ ------------------------------
requests>=2.32,<3
beautifulsoup4>=4.13,<5
scholarly>=1.7.11,<2
# Constraint on a transitive dep, not a direct import. scholarly 1.7.11 asks
# for `bibtexparser` unpinned but imports `bibtexparser.bibdatabase`, which
# bibtexparser 2.x removed -- so an unconstrained resolve installs 2.x and
# `import scholarly` dies with ModuleNotFoundError. The old freeze only worked
# because it happened to pin 1.4.3. Remove this when scholarly supports 2.x.
bibtexparser>=1.4,<2
# web_scrape.py does `import fitz`, which is PyMuPDF. A separate, unrelated
# package literally named `fitz` exists on PyPI -- installing that one is the
# documented mistake, not the fix.
PyMuPDF>=1.24,<2
# PyPDF2 is end-of-life upstream (succeeded by `pypdf`). Kept because the code
# does `import PyPDF2`; switching is a code change, not a pin change.
PyPDF2>=3.0.1,<4

# --- Analysis and figures -- legacy/analysis/ -------------------------------
pandas>=2.2,<4
numpy>=1.26,<3
matplotlib>=3.8,<4
seaborn>=0.13,<0.14
plotly>=5.20,<8
jsonlines>=4,<5
# nltk fetches its own corpora at runtime (punkt, stopwords, POS tagger) via

nltk>=3.9,<4
wordcloud>=1.9.3,<2
